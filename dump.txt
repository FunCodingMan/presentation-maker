import { useEffect } from 'react'
import { undo, redo, getState } from '../../../editor.js'

function useHistoryShortcuts(): void {
    useEffect(() => {
        const handler = (e: KeyboardEvent) => {
            if (getState()?.isPreviewMode) return
            const active = document.activeElement as HTMLElement | null
            const isTyping =
                active?.hasAttribute('contenteditable') ||
                active?.tagName === 'INPUT' ||
                active?.tagName === 'TEXTAREA'

            if (isTyping) return

            const isMod = e.ctrlKey
            if (!isMod) return

            const key = e.key.toLowerCase()
            if (key === 'z') {
                e.preventDefault()
                undo()
            } else if (key === 'y') {
                e.preventDefault()
                redo()
            }
        }

        window.addEventListener('keydown', handler)
        return () => window.removeEventListener('keydown', handler)
    }, [])
}

export { useHistoryShortcuts }

import { useState, useEffect, useCallback } from 'react'
import { undo, redo, getCanUndo, getCanRedo, addEditorChangeHandler } from '../../../editor.js'

type HistoryUI = {
    canUndo: boolean
    canRedo: boolean
    onUndo: () => void
    onRedo: () => void
}

function useHistoryUI(): HistoryUI {
    const [canUndo, setCanUndo] = useState(getCanUndo())
    const [canRedo, setCanRedo] = useState(getCanRedo())

    useEffect(() => {
        const sync = () => {
            setCanUndo(getCanUndo())
            setCanRedo(getCanRedo())
        }
        sync()
        const unsubscribe = addEditorChangeHandler(sync)
        return unsubscribe
    }, [])

    const onUndo = useCallback(() => {
        undo()
    }, [])

    const onRedo = useCallback(() => {
        redo()
    }, [])

    return { canUndo, canRedo, onUndo, onRedo }
}

export { useHistoryUI }

import React, { useRef, useEffect } from 'react'
import styles from './TextInput.module.css'

type TextInputProps = {
    value: string
    onChange: (value: string) => void
    className?: string
    placeHolder?: string
    style?: React.CSSProperties
    readonly?: boolean
    canvasMode?: boolean 
}

function TextInput({ 
    value, 
    onChange, 
    className = '', 
    placeHolder = '', 
    style, 
    readonly = false,
    canvasMode = false
}: TextInputProps) {
    const divRef = useRef<HTMLDivElement>(null)

    useEffect(() => {
        const element = divRef.current
        if (element && document.activeElement !== element && element.innerText !== value) {
            element.innerText = value
        }
    }, [value])

    const handleSave = (e: React.SyntheticEvent<HTMLDivElement>) => {
        if (readonly) return
        
        const newText = (e.currentTarget as HTMLElement).innerText || ''

        if (newText !== value) {
            onChange(newText)
        }
    }

    const combinedClassName = [
        styles.contentEditableDiv,
        canvasMode ? styles.canvasMode : '',
        className
    ].filter(Boolean).join(' ')

    return (
        <div
            ref={divRef}
            contentEditable={!readonly}
            className={combinedClassName}
            style={{
                ...style,
                cursor: readonly ? 'default' : 'text',
            }}
            data-placeholder={readonly ? '' : placeHolder}
            onBlur={handleSave}
            onKeyDown={(e) => {
                if (readonly) return
                if (e.key === 'Escape') {
                    e.preventDefault()
                    if (divRef.current) {
                        divRef.current.innerText = value
                    }
                    e.currentTarget.blur()
                } else if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault()
                    e.currentTarget.blur()
                }
            }}
        >
            {value}
        </div>
    )
}

export { TextInput }

import { useEffect } from 'react'

type UsePreviewNavigationProps = {
    onNextSlide: () => void
    onPrevSlide: () => void
    onClose: () => void
}

export function usePreviewNavigation({ onNextSlide, onPrevSlide, onClose }: UsePreviewNavigationProps) {
    useEffect(() => {
        const handleKeyDown = (e: KeyboardEvent) => {
            if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'Enter') {
                onNextSlide()
            } else if (e.key === 'ArrowLeft') {
                onPrevSlide()
            } else if (e.key === 'Escape') {
                onClose()
            }
        }

        const handleFullscreenChange = () => {
            if (!document.fullscreenElement) {
                onClose()
            }
        }

        window.addEventListener('keydown', handleKeyDown)
        document.addEventListener('fullscreenchange', handleFullscreenChange)
        
        return () => {
            window.removeEventListener('keydown', handleKeyDown)
            document.removeEventListener('fullscreenchange', handleFullscreenChange)
        }
    }, [onNextSlide, onPrevSlide, onClose])
}

import { useState, useEffect } from 'react'

export function usePreviewScale() {
    const [scale, setScale] = useState(1)

    useEffect(() => {
        const calculateScale = () => {
            const windowRatio = window.innerWidth / window.innerHeight
            const slideRatio = 1000 / 562
            
            if (windowRatio > slideRatio) {
                setScale(window.innerHeight / 562)
            } else {
                setScale(window.innerWidth / 1000)
            }
        }

        calculateScale()
        window.addEventListener('resize', calculateScale)
        return () => window.removeEventListener('resize', calculateScale)
    }, [])

    return scale
}

import type { SlideObject, TextObject, ImageObject, FigureObject } from '../../model/types/objects.js'
import { TextInput } from '../common/textInput/TextInput.js'
import { dispatch } from '../../editor.js'
import { modifySlide } from '../../model/functions/presentation.js'
import { updateTextContent } from '../../model/functions/objects.js'
import styles from './SlideObject.module.css'
import { modifyPresentation } from '../../viewmodel/functions/navigation.js'
import React from 'react'

type SlideObjectProps = {
    object: SlideObject
    slideId: string
    readonly?: boolean
    isSelected?: boolean
    onSelect?: (objectId: string, isShiftPressed: boolean) => void
    onDragPointerDown?: (e: React.PointerEvent, objectId: string) => void
    dragOffset?: { dx: number; dy: number }
    onResizePointerDown?: (e: React.PointerEvent, objectId: string) => void
    resizeOffset?: { dw: number; dh: number }
}

type TextObjectProps = { object: TextObject; slideId: string; readonly?: boolean }
type ImageObjectProps = { object: ImageObject }
type FigureObjectProps = { object: FigureObject }

function TextObjectComponent({ object, slideId, readonly }: TextObjectProps) {
    const plainText = object.spans.map(span => span.text).join('')
    const baseStyle = object.spans[0]?.style
    const styleArray = baseStyle?.fontStyle || []

    const decorations: string[] = []
    if (styleArray.includes('underline')) decorations.push('underline')
    if (styleArray.includes('strikethrough')) decorations.push('line-through')

    const textObjectStyle: React.CSSProperties = {
        textAlign: object.textLayout,
        fontFamily: baseStyle?.fontFamily,
        fontSize: `${baseStyle?.fontSize || 24}px`,
        color: baseStyle?.fontColor || '#000000',
        fontWeight: styleArray.includes('bold') ? 'bold' : 'normal',
        fontStyle: styleArray.includes('italic') ? 'italic' : 'normal',
        textDecorationLine: decorations.length > 0 ? decorations.join(' ') : 'none',
    }

    const handleTextChange = (newText: string) => {
        dispatch(modifyPresentation, {
            operation: modifySlide,
            args: {
                slideId,
                operation: updateTextContent,
                args: { objectId: object.id, newText },
            },
        })
    }

    return (
        <TextInput
            value={plainText}
            onChange={handleTextChange}
            style={textObjectStyle}
            readonly={readonly}
            placeHolder="Введите текст"
            className={styles.textObject}
            canvasMode={true}
        />
    )
}

function ImageObjectComponent({ object }: ImageObjectProps) {
    let filterStr = 'none'
    if (object.filters) {
        const blur = object.filters.blur ? `blur(${object.filters.blur}px)` : ''
        const brightness = object.filters.brightness !== undefined ? `brightness(${object.filters.brightness}%)` : ''
        filterStr = `${blur} ${brightness}`.trim() || 'none'
    }
    return (
        <img
            src={object.src}
            alt="Image"
            style={{ filter: filterStr }}
            className={styles.imageObject}
            draggable={false}
        />
    )
}

function FigureObjectComponent({ object }: FigureObjectProps) {
    const { shape, fillColor, strokeColor, strokeWidth } = object.figureStyle
    return (
        <svg className={styles.figureObject} viewBox="0 0 100 100" preserveAspectRatio="none">
            {shape === 'rectangle' && (
                <rect width="100" height="100" fill={fillColor} stroke={strokeColor} strokeWidth={strokeWidth} />
            )}
            {shape === 'circle' && (
                <ellipse cx="50" cy="50" rx="47" ry="47" fill={fillColor} stroke={strokeColor} strokeWidth={strokeWidth} />
            )}
            {shape === 'triangle' && (
                <polygon points="50,0 100,100 0,100" fill={fillColor} stroke={strokeColor} strokeWidth={strokeWidth} />
            )}
        </svg>
    )
}

function SlideObject({ object, slideId, readonly, isSelected, onSelect, onDragPointerDown, dragOffset = { dx: 0, dy: 0 }, onResizePointerDown, resizeOffset = { dw: 0, dh: 0 } }: SlideObjectProps) {
    const handlePointerDown = (e: React.PointerEvent) => {
        if (e.defaultPrevented || readonly) return

        e.currentTarget.setPointerCapture(e.pointerId)

        if (e.shiftKey && isSelected) {
            e.preventDefault()
            onSelect?.(object.id, true)
            return
        }

        if (onSelect && !isSelected) {
            onSelect(object.id, e.shiftKey)
        }

        const target = e.target as HTMLElement
        const isClickInsideText = target.closest('[contenteditable="true"]') !== null

        if (onDragPointerDown && (!isClickInsideText || !isSelected)) {
            e.preventDefault()
            onDragPointerDown(e, object.id)
        }
    }

    const handleClick = (e: React.MouseEvent) => {
        if (readonly) return
        if (onSelect && isSelected && !e.shiftKey) {
            onSelect(object.id, false)
        }
    }
    //TODO: сделать изменение размера через scale transform
    //TODO: синяя кайма не должна увеличиваться вместе с объектом
    const isText = object.type === 'text'
    const scaleX = object.size.width > 0 ? (object.size.width + resizeOffset.dw) / object.size.width : 1
    const scaleY = object.size.height > 0 ? (object.size.height + resizeOffset.dh) / object.size.height : 1

    const dynamicWrapperStyle: React.CSSProperties = {
        left: `${object.position.x}px`,
        top: `${object.position.y}px`,
        width: isText ? `${object.size.width + resizeOffset.dw}px` : `${object.size.width}px`,
        transform: (dragOffset.dx !== 0 || dragOffset.dy !== 0) ? `translate(${dragOffset.dx}px, ${dragOffset.dy}px)` : 'none',
        ...(isText ? { minHeight: `${object.size.height + resizeOffset.dh}px` } : { height: `${object.size.height}px` }),
    }

    const contentScaleStyle: React.CSSProperties = {
        width: '100%',
        height: '100%',
        transformOrigin: 'top left',
        transform: (!isText && (scaleX !== 1 || scaleY !== 1)) ? `scale(${scaleX}, ${scaleY})` : 'none',
    }

    const selectionBoxStyle: React.CSSProperties = {
        width: isText ? '100%' : `${object.size.width + resizeOffset.dw}px`,
        height: isText ? '100%' : `${object.size.height + resizeOffset.dh}px`,
    }
    
    const wrapperClass = `${styles.objectWrapper} ${isSelected ? styles.objectWrapperSelected : ''}`

    return (
        <div onPointerDown={handlePointerDown} onClick={handleClick} className={wrapperClass} style={dynamicWrapperStyle}>
            <div style={contentScaleStyle}>
                {object.type === 'text' && <TextObjectComponent object={object} slideId={slideId} readonly={readonly} />}
                {object.type === 'image' && <ImageObjectComponent object={object} />}
                {object.type === 'figure' && <FigureObjectComponent object={object} />}
            </div>

            {isSelected && !readonly && (
                <div className={styles.selectionBox} style={selectionBoxStyle}>
                    <div
                        className={styles.resizeHandle}
                        onPointerDown={(e) => {
                            e.preventDefault()
                            e.stopPropagation()
                            e.currentTarget.setPointerCapture(e.pointerId)
                            onResizePointerDown?.(e, object.id)
                        }}
                    />
                </div>
            )}
        </div>
    )
}

import type { Slide } from '../../model/types/slide.js'
import styles from './SlideView.module.css'
import { SlideObject } from '../slideObject/SlideObject.js'
import { useObjectDrag } from '../workspace/customHooks/useObjectDrag.js'
import { useObjectResize } from '../workspace/customHooks/useObjectResize.js'


type SlideViewProps = {
    slide: Slide
    readonly?: boolean
    selectedObjectIds?: string[]
    onObjectSelect?: (objectId: string, isShiftPressed: boolean) => void
    id?: string
}

function SlideView({
    slide,
    readonly = false,
    selectedObjectIds = [],
    onObjectSelect,
    id,
}: SlideViewProps) {
    const { handleDragPointerDown, dragOffset } = useObjectDrag({
        selectedObjectIds,
        activeSlideId: slide.id,
        objects: slide.objects,
    })

    const { handleResizePointerDown, resizeOffset } = useObjectResize({
        selectedObjectIds,
        activeSlideId: slide.id,
        objects: slide.objects,
    })
    

    let backgroundStyles: React.CSSProperties = {}

    switch (slide.background.type) {
        case 'color':
            backgroundStyles.backgroundColor = slide.background.color
            break
        case 'image':
            backgroundStyles.backgroundImage = `url(${slide.background.src})`
            backgroundStyles.backgroundSize = 'cover'
            backgroundStyles.backgroundPosition = 'center'
            break
        case 'gradient':
            const angle = slide.background.angle ?? 90
            const colorsStr = slide.background.colors.join(', ')
            
            if (slide.background.gradientType === 'radial') {
                backgroundStyles.background = `radial-gradient(circle, ${colorsStr})`
            } else {
                backgroundStyles.background = `linear-gradient(${angle}deg, ${colorsStr})`
            }
            break
        case 'none':
        default:
            backgroundStyles.backgroundColor = '#ffffff'
            break            
    }

    return (

import { useCallback, useState } from 'react'
import { createRoot, type Root } from 'react-dom/client'
import { flushSync } from 'react-dom'
import { jsPDF } from 'jspdf'
import html2canvas from 'html2canvas'
import { SlideView } from '../../slideView/SlideView.js'
import type { Presentation } from '../../../model/types/presentation.js'
import type { Slide } from '../../../model/types/slide.js'

const EXPORT_SIZE = {
    width: 1000,
    height: 562,
} as const

function createRenderStage(): HTMLDivElement {
    const stage = document.createElement('div')
    stage.style.position = 'fixed'
    stage.style.left = '-10000px'
    stage.style.top = '0'
    stage.style.width = `${EXPORT_SIZE.width}px`
    stage.style.height = `${EXPORT_SIZE.height}px`
    stage.style.pointerEvents = 'none'
    document.body.appendChild(stage)
    return stage
}

async function waitForImages(container: HTMLElement): Promise<void> {
    const images = Array.from(container.querySelectorAll('img'))
    await Promise.all(
        images.map((img) => {
            if (img.complete) return Promise.resolve()
            return new Promise<void>((resolve) => {
                img.onload = () => resolve()
                img.onerror = () => resolve()
            })
        })
    )
}

async function captureSlide(root: Root, stage: HTMLDivElement, slide: Slide): Promise<HTMLCanvasElement> {
    flushSync(() => {
        root.render(<SlideView slide={slide} readonly />)
    })

    await waitForImages(stage)

    return html2canvas(stage, {
        scale: 2,
        useCORS: true,
        logging: false,
        backgroundColor: '#ffffff',
    })
}

function usePdfExport() {
    const [isExporting, setIsExporting] = useState(false)

    const exportToPdf = useCallback(async (presentation: Presentation) => {
        if (isExporting || presentation.slides.length === 0) return

        setIsExporting(true)

        const stage = createRenderStage()
        const root = createRoot(stage)

        try {
            const doc = new jsPDF({
                orientation: 'landscape',
                unit: 'px',
                format: [EXPORT_SIZE.width, EXPORT_SIZE.height],
                hotfixes: ['px_scaling'],
            })

            for (let index = 0; index < presentation.slides.length; index++) {
                if (index > 0) {
                    doc.addPage([EXPORT_SIZE.width, EXPORT_SIZE.height], 'landscape')
                }

                const canvas = await captureSlide(root, stage, presentation.slides[index])
                doc.addImage(canvas, 'PNG', 0, 0, EXPORT_SIZE.width, EXPORT_SIZE.height)
            }

            const fileName = presentation.name.trim() || 'presentation'
            doc.save(`${fileName}.pdf`)
        } catch (err) {
            console.error('Не удалось экспортировать PDF:', err)
        } finally {
            root.unmount()
            stage.remove()
            setIsExporting(false)
        }
    }, [isExporting])

    return { exportToPdf, isExporting }
}

import { useEffect, useRef, useState, useCallback } from 'react'
import { dispatch } from '../../../editor.js'

import { SlideObject } from '../../../model/types/objects.js'
import { Point } from '../../../model/types/objects.js'

type ObjectPositionUpdate = { objectId: string; position: Point }

type UseObjectDragProps = {
    selectedObjectIds: string[]
    activeSlideId: string | null
    objects: SlideObject[]
}

function useObjectDrag({ selectedObjectIds, activeSlideId, objects }: UseObjectDragProps) {
    const [dragOffset, setDragOffset] = useState<{ dx: number; dy: number }>({ dx: 0, dy: 0 })
    const dragRef = useRef<{
        isDragging: boolean
        startMouseX: number
        startMouseY: number
        currentDx: number
        currentDy: number
        initialPositions: Map<string, Point>
    }>({
        isDragging: false,
        startMouseX: 0,
        startMouseY: 0,
        currentDx: 0,
        currentDy: 0,
        initialPositions: new Map(),
    })

    const handleDragPointerDown = useCallback((e: React.PointerEvent, objectId: string) => {
        if (e.button !== 0 || !activeSlideId) return

        let idsToMove: string[]

        if (selectedObjectIds.includes(objectId)) {
            idsToMove = selectedObjectIds
        } else if (e.shiftKey) {
            idsToMove = [...selectedObjectIds, objectId]
        } else {
            idsToMove = [objectId]
        }

        const initialPositions = new Map<string, Point>()
        objects.forEach(obj => {
            if (idsToMove.includes(obj.id)) {
                initialPositions.set(obj.id, obj.position)
            }
        })

        dragRef.current = {
            isDragging: true,
            startMouseX: e.clientX,
            startMouseY: e.clientY,
            currentDx: 0,
            currentDy: 0,
            initialPositions,
        }

        setDragOffset({ dx: 0, dy: 0 })
    }, [selectedObjectIds, activeSlideId, objects])

    useEffect(() => {
        const handlePointerMove = (e: PointerEvent) => {
            if (!dragRef.current.isDragging || !activeSlideId) return

            const dx = e.clientX - dragRef.current.startMouseX
            const dy = e.clientY - dragRef.current.startMouseY
            
            dragRef.current.currentDx = dx
            dragRef.current.currentDy = dy
            setDragOffset({ dx, dy })
        }

        const handlePointerUp = () => {
            const { isDragging, currentDx, currentDy, initialPositions } = dragRef.current

            if (!isDragging || !activeSlideId) return;

            if (Math.abs(currentDx) >= 2 || Math.abs(currentDy) >= 2) {
                const updates: ObjectPositionUpdate[] = []
                
                initialPositions.forEach((pos, objectId) => {
                    updates.push({
                        objectId,
                        position: { x: pos.x + currentDx, y: pos.y + currentDy },
                    })
                })

                dispatch(modifyPresentation, {
                    operation: modifySlide,
                    args: {
                        slideId: activeSlideId,
                        operation: moveObjects,
                        args: { updates },
                    },
                })
            }

            dragRef.current.isDragging = false
            setDragOffset({ dx: 0, dy: 0 })
        }

        document.addEventListener('pointermove', handlePointerMove)
        document.addEventListener('pointerup', handlePointerUp)
        return () => {
            document.removeEventListener('pointermove', handlePointerMove)
            document.removeEventListener('pointerup', handlePointerUp)
        }
    }, [activeSlideId])

    return { handleDragPointerDown, dragOffset }
}

        document.addEventListener('pointermove', handlePointerMove)
        document.addEventListener('pointerup', handlePointerUp)
        return () => {
            document.removeEventListener('pointermove', handlePointerMove)
            document.removeEventListener('pointerup', handlePointerUp)
        }
    }, [activeSlideId])

    return { handleResizePointerDown, resizeOffset }
}

import { useState, useEffect } from 'react'
import { SlideObject } from '../../../model/types/objects'

type UseSelectionProps = {
    activeSlideId: string | null,
}

function useSelection({ activeSlideId }: UseSelectionProps) {
    const [selectedObjectIds, setSelectedObjectIds] = useState<string[]>([])

    useEffect(() => {
        setSelectedObjectIds([])
    }, [activeSlideId])
     
    const handleSelect = (objectId: string, isShiftPressed: boolean, ) => {
        if (isShiftPressed) {
            setSelectedObjectIds(prev =>
                prev.includes(objectId)
                    ? prev.filter(id => id !== objectId)
                    : [...prev, objectId]
            )
        } else {
            setSelectedObjectIds([objectId])
        }
    }

    const clearSelection = () => {
        if (document.activeElement instanceof HTMLElement) {
            document.activeElement.blur()
        }
        setSelectedObjectIds([])
    }

    return { selectedObjectIds, handleSelect, clearSelection }
}

    useSelection

const DEFAULT_STYLE: TextStyle = {
    fontFamily: 'Arial',
    fontSize: 24,
    fontColor: '#000000',
    fontStyle: [],
}

const MIN_FONT_SIZE = 8
const MAX_FONT_SIZE = 200

type FontStyle = TextStyle['fontStyle'][number]

type UseTextStylePanelProps = {
    selectedObjectIds: string[]
    activeSlideId: string | null
    objects: SlideObject[]
}

function useTextStylePanel({ selectedObjectIds, activeSlideId, objects }: UseTextStylePanelProps) {
    const colorInputRef = useRef<HTMLInputElement>(null)

    const foundObj = activeSlideId && selectedObjectIds.length === 1
        ? objects.find(o => o.id === selectedObjectIds[0])
        : null
    const selectedTextObject = foundObj?.type === 'text' ? foundObj : null

    const objectId = selectedTextObject?.id ?? null
    const style = selectedTextObject?.spans[0]?.style ?? DEFAULT_STYLE

    const [fontSizeInput, setFontSizeInput] = useState(String(style.fontSize))
    const [fontColor, setFontColor] = useState(style.fontColor)

    useEffect(() => {
        setFontSizeInput(String(style.fontSize))
        setFontColor(style.fontColor)
    }, [objectId, style.fontSize, style.fontColor])

    const applyStyle = (updates: Partial<TextStyle>) => {
        if (!objectId || !activeSlideId) return
        dispatch(modifyPresentation, {
            operation: modifySlide,
            args: {
                slideId: activeSlideId,
                operation: updateTextObjectStyle,
                args: { objectId, newTextStyle: { ...style, ...updates } },
            },
        })
    }

    useEffect(() => {
        const input = colorInputRef.current
        if (!input || !objectId || !activeSlideId) return

        const handleNativeColorChange = (e: Event) => {
            if (!(e.target instanceof HTMLInputElement)) return
            if (e.target.value !== style.fontColor) {
                applyStyle({ fontColor: e.target.value })
            }
        }

        input.addEventListener('change', handleNativeColorChange)
        return () => input.removeEventListener('change', handleNativeColorChange)
    }, [objectId, activeSlideId, style])

    const commitFontSize = (value: number) => {
        const clamped = Math.max(MIN_FONT_SIZE, Math.min(MAX_FONT_SIZE, value))
        setFontSizeInput(String(clamped))
        if (clamped !== style.fontSize) {
            applyStyle({ fontSize: clamped })
        }
    }

    const handleFontSizeInputChange = (raw: string) => {
        setFontSizeInput(raw)
    }

    const handleFontSizeBlur = () => {
        const parsed = Number(fontSizeInput)
        if (fontSizeInput.trim() === '' || Number.isNaN(parsed)) {
            setFontSizeInput(String(style.fontSize))
            return
        }
        commitFontSize(parsed)
    }

    const handleFontSizeKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
        if (e.key === 'Enter') {
            e.currentTarget.blur()
        }
    }

    const handleFontColorChange = (color: string) => {
        setFontColor(color)
    }

    const handleFontFamilyChange = (fontFamily: string) => applyStyle({ fontFamily })
    const handleFontSizeStep = (delta: number) => commitFontSize(style.fontSize + delta)
    const handleFontStyleToggle = (item: FontStyle) => {
        const fontStyle = style.fontStyle.includes(item)
            ? style.fontStyle.filter(s => s !== item)
            : [...style.fontStyle, item]
        applyStyle({ fontStyle })
    }

    return {
        isVisible: selectedTextObject !== null,
        colorInputRef,
        fontFamily: style.fontFamily,
        fontSizeInput,
        fontColor,
        fontStyle: style.fontStyle,
        handleFontFamilyChange,
        handleFontSizeInputChange,
        handleFontSizeBlur,
        handleFontSizeKeyDown,
        handleFontSizeStep,
        handleFontColorChange,
        handleFontStyleToggle,
    }
}

export type { FontStyle }

import { useEffect } from 'react'
import { dispatch } from '../../../editor.js'
import { modifyPresentation } from '../../../viewmodel/functions/navigation.js'
import { modifySlide } from '../../../model/functions/presentation.js'
import { removeObjects, bringToFront, sendToBack } from '../../../model/functions/objects.js'

type UseWorkspaceShortcutsProps = {
    selectedObjectIds: string[]
    activeSlideId: string | null
    onDelete: () => void
}

export function useWorkspaceShortcuts({ selectedObjectIds, activeSlideId, onDelete }: UseWorkspaceShortcutsProps) {
    useEffect(() => {
        const handleKeyDown = (e: KeyboardEvent) => {
            const activeElement = document.activeElement
            const isTyping = activeElement?.hasAttribute('contenteditable')
            if (isTyping) return

            if ((e.key === 'Delete' || e.key === 'Backspace') && selectedObjectIds.length > 0 && activeSlideId) {
                dispatch(modifyPresentation, {
                    operation: modifySlide,
                    args: {
                        slideId: activeSlideId,
                        operation: removeObjects,
                        args: { objectIds: selectedObjectIds },
                    },
                })
                onDelete()
            }

            if ((e.key === ']' || e.key === 'ArrowUp') && e.ctrlKey && selectedObjectIds.length > 0 && activeSlideId) {
                e.preventDefault()
                dispatch(modifyPresentation, {
                    operation: modifySlide,
                    args: {
                        slideId: activeSlideId,
                        operation: bringToFront,
                        args: { objectId: selectedObjectIds[0] },
                    },
                })
            }

            if ((e.key === '[' || e.key === 'ArrowDown') && e.ctrlKey && selectedObjectIds.length > 0 && activeSlideId) {
                e.preventDefault()
                dispatch(modifyPresentation, {
                    operation: modifySlide,
                    args: {
                        slideId: activeSlideId,
                        operation: sendToBack,
                        args: { objectId: selectedObjectIds[0] },
                    },
                })
            }
        }

        window.addEventListener('keydown', handleKeyDown)
        return () => window.removeEventListener('keydown', handleKeyDown)
    }, [selectedObjectIds, activeSlideId, onDelete])
}
